// Time tracking for subsystem dynamic workload drift let subsystemOptTimestamps = { gpu: 0, ram: 0, net: 0, disk: 0, cpu: 0 }; function toggleSingleSubsystem(id) { subsystemOptimizedState[id] = true; subsystemOptTimestamps[id] = Date.now() / 1000.0; userAchievements.optimizer = true; updateTelemetry(); } function triggerMasterEcoOptimizer() { const nowSec = Date.now() / 1000.0; Object.keys(subsystemOptimizedState).forEach(k => { subsystemOptimizedState[k] = true; subsystemOptTimestamps[k] = nowSec; }); userAchievements.optimizer = true; fetch(`${API}/eco-optimize`, { method: 'POST' }) .then(r => r.json()) .then(data => { alert("🌱 " + data.message); updateTelemetry(); }); } async function updateTelemetry() { try { const grid = document.getElementById('gridSelect').value; const res = await fetch(`${API}/telemetry`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ grid: grid, subsystems: subsystemOptimizedState, opt_timestamps: subsystemOptTimestamps }) }); const data = await res.json(); lastTelemetryData = data; // Reset state for subsystems that drifted back to active const nowSec = Date.now() / 1000.0; Object.keys(subsystemOptimizedState).forEach(k => { if (nowSec - subsystemOptTimestamps[k] >= 20.0) { subsystemOptimizedState[k] = false; } }); document.getElementById('scoreVal').innerText = data.sustainability_score; document.getElementById('scoreBar').style.width = `${data.sustainability_score}%`; document.getElementById('cpuVal').innerText = `${data.cpu_percent}%`; document.getElementById('wattsVal').innerText = data.current_watts; document.getElementById('co2Val').innerText = data.co2_grams; // Update all UI components... renderSubsystems(data.subsystem_details, data.grid_factor || 710.0); updateAdvisor(data.cpu_percent, data.sustainability_score, data.unopt_count); updateEcoBattleUI(data.current_watts, data.grid_factor || 710.0); // ... remaining telemetry logic } catch (err) {} }
